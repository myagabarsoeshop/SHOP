function renderTable(list) { const tbody = document.getElementById("sr-table-body"); if (!tbody) return; tbody.innerHTML = ""; // ID နှစ်မျိုးလုံးကို ရှာဖွေနိုင်အောင် ထားရှိပေးထားသည် const tfoot = document.getElementById("sr-table-foot") || document.getElementById("sr-table-footer"); if (list.length === 0) { tbody.innerHTML = `စတော့မှတ်တမ်း မတွေ့ရှိပါ`; if (tfoot) tfoot.innerHTML = ""; return; } const monthlyTotals = Array(13).fill(0); let sumIn = 0; let sumOut = 0; let sumBal = 0; let sumVal = 0; list.forEach((item, index) => { const tr = document.createElement("tr"); tr.className = "hover:bg-slate-50 dark:hover:bg-slate-900/50 transition border-b border-slate-100 dark:border-slate-800/60"; const currentBal = Number(item.current_balance || 0); const costPrice = Number(item.cost_price || 0); const valuation = currentBal * costPrice; const totalIn = Number(item.total_in || 0); const totalOut = Number(item.total_out || 0); sumIn += totalIn; sumOut += totalOut; sumBal += currentBal; sumVal += valuation; let rowHtml = ` ${index + 1} ${escapeHtml(item.category || "-")}
${escapeHtml(item.item_name)}
${escapeHtml(item.sku || "-")}
`; for (let m = 1; m <= 12; m++) { const mQty = Number(item.monthly_balances?.[m] ?? item[`m${m}`] ?? 0); monthlyTotals[m] += mQty; rowHtml += ` ${mQty.toLocaleString()} `; } rowHtml += ` +${totalIn.toLocaleString()} -${totalOut.toLocaleString()} ${currentBal.toLocaleString()} ${valuation.toLocaleString()} Ks `; tr.innerHTML = rowHtml; tbody.appendChild(tr); }); // အောက်ဆုံး Total Footer ကို လအလိုက် တန်ဖိုးများနှင့်တကွ တိုက်ရိုက် ထည့်သွင်းခြင်း if (tfoot) { let footHtml = ` Total Summary `; for (let m = 1; m <= 12; m++) { footHtml += ` ${monthlyTotals[m].toLocaleString()} `; } footHtml += ` +${sumIn.toLocaleString()} -${sumOut.toLocaleString()} ${sumBal.toLocaleString()} ${sumVal.toLocaleString()} Ks `; tfoot.innerHTML = footHtml; } }
